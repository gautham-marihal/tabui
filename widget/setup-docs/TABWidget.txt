import React from "react";
import Image from "next/image";

export default function TABWidget() {
  const [showTABWidget, setShowTABWidget] = React.useState(false);

  React.useEffect(() => {
    //@ts-expect-error - TODO:. Fix this. Adding this comment to bypass vercel deploy blocker

          position: "fixed",

          bottom: 30,
          display: "grid",
          placeItems: "center",

        }}
      >
        &nbsp;
      </div>

      <div
        id="tab-icon"

        style={{
          backgroundColor: "#732b90",
          borderRadius: "50%",
